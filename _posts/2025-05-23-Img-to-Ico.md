from PIL import Image
import io
'''
# 脚本用来将PNG图片转换为ICO格式
# 用法: python png_to_ico.py <png_path> <ico_path>
# 例如: python png_to_ico.py image.png image.ico
# 注意: 请确保已经安装了Pillow库
'''
def png_to_ico(png_path, ico_path):
    """
    将PNG图片转换为ICO格式
    :param png_path: 图片路径
    :param ico_path: 输出ICO图片路径
    :param sizes: ICO文件中包含的图标尺寸列表
    """
    sizes = [(16, 16), (24, 24), (32, 32), (48, 48), (64, 64), (128, 128), (256, 256)]
    img = Image.open(png_path)
    # 确保图片有 RGBA 模式（支持透明度）
    if img.mode != 'RGBA':
        img = img.convert('RGBA')
    # img.save(ico_path, format='ICO', sizes=sizes) 
    
    # 创建不同尺寸的图标，使用高质量重采样
    # icon_sizes = [img.resize(size, Image.BICUBIC) for size in sizes]
    icon_sizes = [img.resize(size, Image.Resampling.LANCZOS) for size in sizes]

    # 保存ICO文件，使用所有尺寸版本
    icon_sizes[0].save(
        ico_path, 
        format='ICO',
        append = True,
        append_images=icon_sizes[1:],
        sizes=sizes
    )

def selectDir():
    """
    选择文件夹并处理其中的所有PNG文件
    """
    import tkinter as tk
    from tkinter import filedialog
    import os
    
    root = tk.Tk()
    root.withdraw()  # 隐藏主窗口
    
    # 选择文件夹
    folder_path = filedialog.askdirectory()
    if not folder_path:
        return
    
    # 遍历文件夹中的所有文件
    for filename in os.listdir(folder_path):
        if filename.lower().endswith('.png'):
            png_path = os.path.join(folder_path, filename)
            # 生成对应的ico文件名
            ico_path = os.path.join(folder_path, os.path.splitext(filename)[0] + '.ico')
            try:
                png_to_ico(png_path, ico_path)
                print(f'已转换: {filename}')
            except Exception as e:
                print(f'转换失败 {filename}: {str(e)}')

if __name__ == '__main__':
    selectDir()
