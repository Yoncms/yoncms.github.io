import time
import json

from concurrent.futures import ThreadPoolExecutor
import threading
import queue

# 改进了5_seeTv.py的代码，修改了att和moduleId的传递，并且把
# get_lectures函数进行了拆分。
# 需要access_token_，为了方便再手机上运行，所以通过手动获取。
# Version：25.07.04

class hxSee(Req):
    def __init__(self, token):
        super().__init__()
        self.token = token
        self.host = 'fj.rcpxpt.com'
        self.upHead({'Host': self.host})
        self.upCook({'account_token_': token})
        self.zhangData = []

    def get_tasks(self, cpid):
        self.cpid = cpid
        url = f'classPackage/findRequiredCourse/{cpid}'
        res = self.query(url, {})
        with ThreadPoolExecutor(max_workers=10) as executor:
            for n, course in enumerate(res):
                percent = course['studyPercent']
                print(f"⌛ 正在获取课程 {n+1}/{len(res)} - {course['name']}")
                if percent != '100.00':
                    executor.submit(self.get_modules, course)
                else:
                    print(f"🎉 课程 {course['name']} 已完成")
                time.sleep(0.5)
                
    def get_modules(self, course):
        class_id = course['id']
        commodity_id = course['commodityId']
        data = {
            'page': 1, 'classtypeId': class_id, 'commodityId': commodity_id, 'videoIsBuy': 0
        }
        url = 'commoditynew/findPcLectrueById'
        res = self.query(url, data)['data']
        for i, mod in enumerate(res):
            moduleId = mod['moduleId']
            lData = (class_id, mod, moduleId)
            self.get_lectures(lData)

    def get_lectures(self, lData):
        class_id, mod, moduleId = lData
        lectures = self._fetch_lectures(class_id, mod['id'])      # 拉取列表
        for lec in lectures:
            print(lec)
            self._process_lecture(lec, class_id, moduleId)  

    # 获取课程里每个讲座的数据
    def _fetch_lectures(self, class_id, chapter_id):
        data = {
            'page': 1,
            'pageSize': 100,
            'classtypeId': class_id,
            'chapterId': chapter_id,
            'videoIsBuy': 0,
            'classPackageId': self.cpid
        }
        url = 'commoditynew/queryLecturesByChapterId'
        return self.query(url, data).get('data', [])

    # 生成单个视频的相关信息
    def _process_lecture(self, lec, class_id, moduleId):
        if lec.get('videoLeanPercent') == '100':
            return
        video = lec.get('video')
        if not video:
            return
        h, m, s = map(int, video['videoTime'].split(':'))
        # 视频的总时长
        total_seconds = h * 3600 + m * 60 + s
        # 视频还需要学习的时长，就是剩余的时长
        left = int(total_seconds * (100 - int(lec.get('videoLeanPercent') or 0)) / 100)
        # 生产uhid和uhaid
        self.addHistory(lec['id'], class_id)
        self._add_to_zhangData(lec, class_id, total_seconds, left, moduleId)
    
    # 生成刷课任务，zhangData是一个数组，元素是一个字典
    def _add_to_zhangData(self, lec, class_id, total_seconds, left, moduleId):
        self.zhangData.append({
            'videoId': lec['videoId'],
            'lectureId': lec['id'],
            'name': lec['lectureName'],
            'studyLength': left,
            'classTypeId': class_id,
            'uhaId': self.uhaid,
            'uhId': self.uhid,
            'played': 0,
            'vcount': total_seconds,
            'moduleId': moduleId
        })

    def addHistory(self, lectureId, classTypeId):
        url = 'userHistory/addHistory'
        data = {
            'lectureId': lectureId,
            'classTypeId': classTypeId,
            'classPackageId': self.cpid
        }
        res = self.query(url, data)
        self.uhid = res.get('uhId')
        self.uhaid = res.get('uhaId')

    def applyLicense(self, item):
        url = 'license/apply/video/license'
        sss = f"{self.scheme}{self.host}/classModule/video/{item['moduleId']}/"
        sss += f"{item['classTypeId']}/{item['lectureId']}/0/0?videoIsBuy=0&pId={self.cpid}" 
        data = {
            'videoId': item['videoId'],
            'videoName': item['name'],
            'videoUrl': sss,
            'classPackageId': self.cpid,
            'classTypeId': item['classTypeId'],
            'json': 1    # 发送请求时如果需要以json格式的数据发送，需要把设置json参数
        }
        self.query(url, data)

    def setTime(self, item):
        self.applyLicense(item)
        url = 'userHistory/setTime'
        att = min(60, max(0, item['vcount'] - item['played']))
        data = {
            'stimes': item['played'],
            'status': 2,
            'uhId': item['uhId'],
            'uhaId': item['uhaId'],
            'actualTime': att
        }
        self.query(url, data)
        print(f"✅ {item['name']} - played: {item['played']}")

    def zbpt(self, item):
        url = 'userHistory/addZbptSupervise'
        data = {
            'packageId': self.cpid,
            'classTypeId': item['classTypeId'],
            'lectureId': item['lectureId'],
            'userHistoryId': item['uhId'],
            'uhaId': item['uhaId'],
            'status': 0,
            'studyLength': item['played']
        }
        self.query(url, data)

    def updateFinal(self, item):
        self.setTime(item)
        url1 = 'userHistory/updateStudyValid'
        url2 = 'userHistory/updateStatus'
        st = item['studyLength']
        self.query(url1, {
            'uhId': item['uhId'],
            'uhaId': item['uhaId'],
            'studyLength': st,
            'actualTime': st
        })
        self.query(url2, {
            'id': item['uhId'],
            'studyStatus': 3
        })
        print(f"🎉 完成 {item['name']}")

    def myRun(self, counts):
        num = 60-counts
        while num >= 0:
            print('...', end='')
            num -= 1
            time.sleep(1)

    def worker(self, q, next_round, lock):
        while True:
            try:
                item = q.get(timeout=1)
            except queue.Empty:
                return

            item['played'] += 60
            self.setTime(item)
            self.zbpt(item)

            if item['played'] >= item['studyLength']:
                self.updateFinal(item)
            else:
                with lock:
                    next_round.append(item)
            q.task_done()

    def run_tasks(self):
        while self.zhangData:
            print(f"⏱ 模拟中，剩余任务：{len(self.zhangData)}")
            next_round = []
            task_queue = queue.Queue()
            lock = threading.Lock()
        
            # 任务入队
            for item in self.zhangData:
                task_queue.put(item)

            threads = []
            thread_count = min(10, task_queue.qsize())  # 最多启动10个线程，或任务数更少

            startTime = time.time()
            for _ in range(thread_count):
                t = threading.Thread(target=self.worker, args=(task_queue, next_round, lock))
                t.start()
                threads.append(t)
                time.sleep(1)

            [t.join() for t in threads]

            endTime = int(time.time() - startTime)
            self.zhangData = next_round
            self.myRun(endTime)
