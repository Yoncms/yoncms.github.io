# easyOCR进行图片识别

<pre>
imgres = request.get(url) # url是图片的网址<!--more-->
# imgres图片不进行保存，而是直接打开进行操作，
# io.BytesIO把bytes数据包装成“文件对象”；
# 就是可以像读写文件一样读写内存中的二进制内容，而不需要保存到硬盘。
img = Image.open(io.BytesIO(imgres)).convert('RGB')
# 转成 numpy array，并把RGB转成BGR
bgr_img = np.array(img)[:, :, ::-1]
# 进行图片识别
reader = easyocr.Reader(['en'])  # 语言可以是任意支持的
results = reader.readtext(
    bgr_img,
    # 如果需要识别的都是字母或数字，可以设置allowlist范围，提高速度识别率
    allowlist='0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ'
    # 如果只有数字的就更简单
    # allowlist='0123456789'
)
res = results[0][1]
