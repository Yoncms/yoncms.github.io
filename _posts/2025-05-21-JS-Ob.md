# Python解除JavaScript最简单的混淆方法
<pre>
import re
pattern = re.compile(r'\b([a-zA-Z0-9]+)\b')<!--more-->
def rep(p, k):
     # format(i, 'b')是字符转二进制
     dx = {format(i, 'b'):k[i] for i in range(len(k))}
     # 得到{'0': 'quan', '1': 'fan', '10': 'zhang', '11': 'rong', '100': 'yi', '101': 'wen', '110': 'chen', '111': 'jun'}
     print(dx)
     res = pattern.sub(lambda match:dx.get(match.group(1), match.group(1)), p)
     print(res)
     return res
     
# p用来指定字符串的顺序和格式
p = "10://0.11.111-110"
# 打乱顺序的字符串
k = 'quan|fan|zhang|rong|yi|wen|chen|jun'.split('|')

rep(p, k)

# 以下是16进制的写法
# 打乱顺序的字符串
sa = 'ew|ccc|qqq|lkj|vd|ax|xb|xc|xd|xe|sd|ad|adf|sd|xx|ww'

# 分割成列表
cd = sa.split('|')

# fromat是将数字转换为十六进制，如果元素再16个以内，计数比较方便
xx = {format(i, 'x'):cd[i] for i in range(len(cd))}
# 得到字典：{'0': 'ew', '1': 'ccc', '2': 'qqq', '3': 'lkj', '4': 'vd', '5': 'a', '6': 'b', '7': 'c', '8': 'd', '9': 'e', 'a': 'sd', 'b': 'ad', 'c': 'adf', 'd': 'sd', 'e': 'xx', 'f': 'ww'}
# print(xx)

# p用来指定字符串的顺序和格式
p = 'f:9:8*d:3:6:2-0-5-e:1*a/b/c'

# 把p字符串里的相关字符用sa字符串里的字符来代替
patter = re.compile(r'[a-f0-9]+')
match = patter.sub(lambda m: xx.get(m.group(), m.group()), p)
print(match)
