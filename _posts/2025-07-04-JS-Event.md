const dv = document.querySelector('#dv');
const aa = document.querySelectorAll('a');
// 把所有子节点的相同的事件，全部绑定到父节点上
dv.addEventListener('click', (e)=&gt;{
    const et = e.target;
    for(let em of aa){
        em.style.color = '#666';
    }
    document.title = et.innerText;
    et.style.color = 'blue';
});


dv.addEventListener('zhang', (e)=&gt;{
    const ed = e.detail;
    for(let em of aa){
        if(em.innerText.includes(ed.user)) {
            em.innerText = ed.name;
        }
    }
    console.log(ed.user, ed.name);
});

const arr = ['aaaa', 'bbbb', 'cccc', 'dddd', 'eeee'];
let i = 0, l=arr.length;
let timer = setInterval(()=&gt;{
    if(i&gt;100) return clearInterval(timer);
    let arri = arr[i % l];
    // 这里就是自定义的事件，事件叫zhang
    dv.dispatchEvent(new CustomEvent('zhang', {
        detail:{'user': arri, 'name': arri + i*10} 
    }));
    i++;
}, 1000);
