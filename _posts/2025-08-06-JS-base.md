// 16
function hexs(hex, ss) {
	if(ss!=16) {
		const bytes = new Uint8Array(hex.match(/[\da-f]{2}/gi).map(h => parseInt(h, 16)));
		return new TextDecoder().decode(bytes);
	}
	return Array.from(new TextEncoder().encode(hex))
		.map(b => b.toString(16).padStart(2, '0'))
		.join('');
}

function hexx(str, ss) {
	if(ss!=36) {
		// 手动实现 base36 转 BigInt
		let bigint = [...str].reduce((acc, char) => {
		let digit = parseInt(char, 36);
		if (isNaN(digit)) throw new Error(`Invalid base36 character: ${char}`);
			return acc * 36n + BigInt(digit);
		}, 0n);

		let hex = bigint.toString(16);
		if (hex.length % 2 !== 0) hex = '0' + hex;

		const bytes = hex.match(/.{2}/g).map(h => parseInt(h, 16));
		return new TextDecoder().decode(new Uint8Array(bytes));
	}
	const bytes = new TextEncoder().encode(str); // UTF-8 编码
	let hex = Array.from(bytes).map(b => b.toString(16).padStart(2, '0')).join('');
	let bigint = BigInt('0x' + hex);
	return bigint.toString(36); // 转为 base36
}

const BASE62_CHARS = Chars();
//62
function bytesToBase62(bytes) {
  let hex = Array.from(bytes).map(b => b.toString(16).padStart(2, '0')).join('');
  let bigint = BigInt('0x' + hex);

  if (bigint === 0n) return '0';

  let result = '';
  while (bigint > 0) {
    let rem = bigint % 62n;
    result = BASE62_CHARS[Number(rem)] + result;
    bigint = bigint / 62n;
  }
  return result;
}
function base62ToBytes(base62Str) {
  let bigint = 0n;
  for (let char of base62Str) {
    let index = BASE62_CHARS.indexOf(char);
    if (index === -1) throw new Error(`Invalid base62 character: ${char}`);
    bigint = bigint * 62n + BigInt(index);
  }

  let hex = bigint.toString(16);
  if (hex.length % 2 !== 0) hex = '0' + hex;

  return new Uint8Array(hex.match(/.{2}/g).map(h => parseInt(h, 16)));
}

function enBase62(str) {
	const bytes = new TextEncoder().encode(str);
	return bytesToBase62(bytes);
}

function deBase62(base62Str) {
	const bytes = base62ToBytes(base62Str);
	return new TextDecoder().decode(bytes);
}

function toBase62(bytes) {
	let hex = Array.from(bytes).map(b => b.toString(16).padStart(2, '0')).join('');
	let bigint = BigInt('0x' + hex);
	if (bigint === 0n) return '0';

		let rem = bigint % 62n;

	}
	return result;
}

// 🔁 base62 解码
function fromBase62(str) {
	let bigint = 0n;
	for (let ch of str) {
		let index = BASE62_CHARS.indexOf(ch);
		if (index === -1) throw new Error(`Invalid base62 character: ${ch}`);
		bigint = bigint * 62n + BigInt(index);
	}
	let hex = bigint.toString(16);
	if (hex.length % 2 !== 0) hex = '0' + hex;
	return new Uint8Array(hex.match(/.{2}/g).map(h => parseInt(h, 16)));
}

// ✅ 简单同步校验和函数（哈希累加）
function simpleChecksum(data, length = 2) {
	let hash = 0;
	for (let i = 0; i < data.length; i++) {
		hash = (hash * 31 + data[i]) % 65536;
	}
	const result = new Uint8Array(length);
	for (let i = 0; i < length; i++) {
		result[i] = (hash >> (8 * i)) & 0xff;
	}
	return result;
}

// ✅ 编码：字符串 → base62（带盐 + 校验）
function encodeWithSaltAndChecksumSync(str, saltLength = 3, checksumLength = 2) {
	const encoder = new TextEncoder();
	const payload = encoder.encode(str);
	const salt = crypto.getRandomValues(new Uint8Array(saltLength));
	//console.log('salt', salt);
	const combined = new Uint8Array(salt.length + payload.length);
	combined.set(salt);
	combined.set(payload, salt.length);

	const checksum = simpleChecksum(combined, checksumLength);

	const final = new Uint8Array(combined.length + checksumLength);
	final.set(combined);
	final.set(checksum, combined.length);

	return toBase62(final);
}

// ✅ 解码：base62 → 原始字符串（验证校验）
function decodeWithSaltAndChecksumSync(base62Str, saltLength = 3, checksumLength = 2) {
	const data = fromBase62(base62Str);
	const contentLen = data.length - checksumLength;

	if (contentLen < saltLength) throw new Error("Invalid data: too short");

	const combined = data.slice(0, contentLen);
	const payload = combined.slice(saltLength);
	const givenChecksum = data.slice(contentLen);
	const expectedChecksum = simpleChecksum(combined, checksumLength);

	for (let i = 0; i < checksumLength; i++) {
		if (givenChecksum[i] !== expectedChecksum[i]) {
			throw new Error("校验失败：数据已被修改或损坏");
		}
	}

	return new TextDecoder().decode(payload);
}
