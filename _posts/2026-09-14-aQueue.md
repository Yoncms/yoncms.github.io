# 封装asyncio的queue队列
<pre>
import asyncio<!--more-->
class Aqueue:
    def __init__(self, maxsize=0):
        self.num = 15
        msize = min(maxsize, self.num * 2)
        self.queue = asyncio.Queue(maxsize=msize)

    async def producer(self, items):
        for item in items:
            await self.queue.put(item)
            print('1', end='', flush=True)
        for _ in range(self.num):
            print('2', end='', flush=True)
            await self.queue.put(None)  # 每个消费者一个哨兵

    async def consumer(self, *kw):
        kw0,kw1,kw2 = kw + (None, None)
        while True:
            item = await self.queue.get()
            if item is None:
                self.queue.task_done()
                break
            try:
                if isinstance(kw0, type):
                    async with kw0(item) as kws:
                        foo = getattr(kws, kw1)
                        if kw2 is not None: await foo(kw2)
                        else: await foo()
                else:
                    if kw1 is not None: await kw0(item, kw1)
                    else: await kw0(item)
            except Exception as e:
                print(e)
            finally:
                self.queue.task_done()
                print('3', end='', flush=True)

    async def worker(self, datas, *kw):
        task1 = asyncio.create_task(self.producer(datas))
        task2 = [asyncio.create_task(self.consumer(*kw)) for _ in range(self.num)]
        await task1
        await self.queue.join()
        for task in task2:
            task.cancel()
            print('4', end='', flush=True)
            # print('■', end='', flush=True)
  
async def xQueue(datas, *kw):
    return await getattr(Aqueue(len(datas)), 'worker')(datas, *kw)
