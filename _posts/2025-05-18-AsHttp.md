import aiohttp
import asyncio
import logging
import re
from fake_useragent import UserAgent

# 支持get和post请求
# 支持设置请求头
# 支持设置请求参数
# 支持设置请求代理

class ashttp:
    scheme = 'https://'
    cook = {}
    respx = 0
    def __init__(self):
        self.hs = {
            'User-Agent': UserAgent().random,
            'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,image/apng,*/*;q=0.8,application/signed-exchange;v=b3;q=0.9',
            'Accept-Language': 'zh-CN,zh;q=0.9'
        }
        self.semaphore = asyncio.Semaphore(5)  # 限制并发数    
        self.space = re.compile(r'&nbsp;|[\s ]+')
        self.session: Optional[aiohttp.ClientSession] = None

    async def __aenter__(self):
        self.session = aiohttp.ClientSession(
            connector=aiohttp.TCPConnector(ssl=False, limit=None),
            timeout=aiohttp.ClientTimeout(
                total=None, sock_connect=None, sock_read=None
            ),
            cookies = self.cook,
            headers=self.hs,
            trust_env=True,
            proxy='http://127.0.0.1:10808'
        )
        return self

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if self.session and not self.session.closed:
            await self.session.close()

    async def query(self, url, data=None, params=None, rh=None):
        session = self.session
        try:
            async with self.semaphore:  # 使用信号量限制并发
                method = session.get if data == 'get' else session.post
                request_kwargs = {'url': url}

                # 如果是POST请求并且data有值
                if data and data != 'get':
                    # 如果是字符串则解析为JSON，否则直接传递data
                    if isinstance(data, str):
                        request_kwargs['json'] = data
                    else:
                        request_kwargs['data'] = data
                # 如果是get请求并且params有值，把值赋给request_kwargs
                elif data == 'get' and params:
                    request_kwargs['params'] = params

                # 执行请求并处理响应
                async with method(**request_kwargs) as response:
                    return await self.process_response(response, rh)

        except Exception as e:
            logging.exception(f"请求出错: {e}, URL: {url}")
            return None     
    
    def setcook(self, cook):
        cks = {}
        # print(cook)
        for x in cook:
            w = re.search(r'(.*?=.*?);', x).group(1)
            if '=' in w:
                em = w.split('=')
                cks[em[0]] = em[1]
        self.cook = cks
        # print(cks)

    async def process_response(self, response, rh=0):
        # print('.......', response.headers.get('Content-Type', ''))
        if (rst:=response.status) >= 400:
            print('httpcode:', rst)
            return None
        try:
            if rh==1:
                self.setcook(response.headers.getall('Set-Cookie'))
                return self.cook
            if rst != 200:
                return None
            content_type = response.headers.get('Content-Type', '')
            # print(content_type)
            if 'text/' in content_type:
                return await response.text()
            elif 'application/json' in content_type:
                return await response.json()
            else:
                if self.respx==0:
                    return await response.read()
                return response
        except Exception as e:
            # logging.exception(f"处理响应出错: {e}")
            return None
