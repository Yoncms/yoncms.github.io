# 动态生成script标签容易踩的坑
<pre>
function injectScript(files) {
    const dby = document.body<!--more--> || document.querySelector('body');
    if(!dby)return;

    function loadscript(index){
        if(index >= files.length)return;
        const script = document.createElement('script');
        script.onload = ()=>{
            // 这里的代码会等到script加载完后才会执行
            script.remove();
            loadscript(index+1);
        };
        // 动态创建script标签时，要先绑定onload事件，再设置src；
        // 也就是要在script绑定完load事件的后面，再设置src，否则
        // 可能因为缓存或网速多块而错过事件触发；
        // 缓存里的脚本也要“加载”，但浏览器可能加载得非常快（几乎
        // 是瞬间），如果你先设置了src，再绑定onload，浏览器可能
        // 在你绑定之前就完成了加载；onload是要等加载完才触发，而在
        // 你还没绑定事件的时候，人家已经加载完了，所以 onload 根本
        // 不会触发。
        script.src = chrome.runtime.getURL(files[index] + '.js');
        dby.appendChild(script);
    }
    loadscript(0);
}

injectScript(['execute', 'scr']);
